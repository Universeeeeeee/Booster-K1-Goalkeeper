cmake_minimum_required(VERSION 3.16)
project(zed_sdk_receiver LANGUAGES CXX)
set(ZED_SDK_ROOT "" CACHE PATH "Isolated unpacked ZED SDK root")
if(NOT ZED_SDK_ROOT)
  message(FATAL_ERROR "Pass -DZED_SDK_ROOT=/absolute/path/to/sdk")
endif()
find_path(ZED_INCLUDE_DIR sl/Camera.hpp HINTS "${ZED_SDK_ROOT}/include" NO_DEFAULT_PATH)
find_library(ZED_LIBRARY sl_zed HINTS "${ZED_SDK_ROOT}/lib" NO_DEFAULT_PATH)
find_path(CUDA_INCLUDE_DIR cuda.h HINTS "$ENV{CUDA_HOME}/include" /usr/local/cuda/include)
if(NOT ZED_INCLUDE_DIR OR NOT ZED_LIBRARY OR NOT CUDA_INCLUDE_DIR)
  message(FATAL_ERROR "Missing SDK library/header or CUDA headers")
endif()
find_package(Threads REQUIRED)
add_executable(zed_sdk_receiver zed_sdk_receiver.cpp)
target_compile_features(zed_sdk_receiver PRIVATE cxx_std_17)
target_compile_options(zed_sdk_receiver PRIVATE -Wall -Wextra)
target_include_directories(zed_sdk_receiver SYSTEM PRIVATE "${ZED_INCLUDE_DIR}" "${CUDA_INCLUDE_DIR}")
target_link_libraries(zed_sdk_receiver PRIVATE "${ZED_LIBRARY}" Threads::Threads)
set_target_properties(zed_sdk_receiver PROPERTIES BUILD_RPATH "${ZED_SDK_ROOT}/lib")
add_executable(prepare_g5_neural prepare_g5_neural.cpp)
target_compile_features(prepare_g5_neural PRIVATE cxx_std_17)
target_include_directories(prepare_g5_neural SYSTEM PRIVATE "${ZED_INCLUDE_DIR}" "${CUDA_INCLUDE_DIR}")
target_link_libraries(prepare_g5_neural PRIVATE "${ZED_LIBRARY}" Threads::Threads)
set_target_properties(prepare_g5_neural PROPERTIES BUILD_RPATH "${ZED_SDK_ROOT}/lib")
